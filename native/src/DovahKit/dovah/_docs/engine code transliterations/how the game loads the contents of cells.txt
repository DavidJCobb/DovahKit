struct FORM {
   char     signature[4];    // 00
   uint32_t length;          // 04
   uint32_t flags;           // 08
   uint32_t form_id;         // 0C
   uint32_t version_control; // 10
   uint16_t form_version;    // 14
   uint16_t vc_version;      // 16
};

// inferred, based on code below. it seems that TESFile actually stores a union 
// of FORM and this
struct GRUP {
   char     signature[4];    // 00
   uint32_t length;          // 04
   char     label[4];        // 08
   uint32_t type;            // 0C
   uint32_t version_control; // 10
   uint16_t form_version;    // 14
   uint16_t vc_version;      // 16
};

struct InteriorCellOffsetTuple {
   uint32_t cell_form_id;      // 00
   uint32_t child_grup_offset; // 04
   uint32_t cell_form_offset;  // 08 // maybe actually the offset of the record body?
};

// Called when loading the `DATA` subrecord
// LE: 4C0960
void TSEObjectCELL::CacheInteriorCellRecordOffset(TESFile* file, uint32_t offset) {
   if (!this->IsInterior())
      return;
   TESFile* edi = file->sub444890();
   if (!edi)
      edi = file;
   if (edi == this->GetSourceFile(0)) {
      // inlined:
      auto* data = this->GetInteriorData(); // this->IsInterior() ? this->loadedData : nullptr
      if (!data)
         return;
      this->loadedData->unk5C = offset;
      return;
   }
   file->StoreCellOffset_B(this, offset);
}

// LE: 445910
void TESFile::StoreCellOffset_B(TESForm* form, uint32_t offset) {
   bool     found = false; // dl
   uint32_t edi   = 0;
   do {
      if (edi >= this->unk430->size()) // size == dword at offset 0x10
         goto reached_end;
      InteriorCellOffsetTuple& item = this->unk430->operator[](esi);
      if (item.cell_form_id == form->formID) {
         item.cell_form_offset = offset;
         found = true;
      }
   } while (++edi, !found);
   return;
reached_end:
   if (found)
      return;
   this->unk430->emplace_back() = { form->formID, 0, offset }; // inlined
}

      return true;

   for(uint32_t i = 0; i < file_count; ++i) { // ebx as loop counter
      TESFile* edi = this->GetSourceFile(i)->sub4472D0(0x4000);
      if (!edi)

         continue;

      if (!ebp)
         ebp = edi;
      
      bool al;
      if (this->IsInterior()) { // this->unk2C & 1
         [[unknown_type]] uint32_t eax;
         if (ebp == this->GetSourceFile(0)) {
            eax = this->parentWorld; // DWORD PTR [ESI+0x84]
         } else {
            eax = edi->OffsetOfInteriorCellChildGroup(this); // documented below
         }

            al = edi->SeekTo(eax); // documented below
         } else {
            al = edi->SeekToRecordOf(this); // documented below
         }
      } else {
         al = edi->SeekToRecordOf(this);
      }
      if (al) {
         if (this->LoadChildrenFromFile(edi)) // documented below
            continue;
      }

   }

      if (!this->IsInterior()) { // !(this->unk2C & 1)

            // Probably promotes "large refs" to persistent.
            //
            this->parentWorld->unk170.sub4AD090(this);
         }
      }

   this->sub4CD5E0();

// LE: 4450D0
uint32_t TESFile::OffsetOfInteriorCellChildGroup(TESObjectCELL* cell) {
   auto* edi = this->unk430; // looks like a BSTArray<T>* maybe
   for(uint32_t i = 0; i < edi->size(); ++i) {
      InteriorCellOffsetTuple& item = edi->operator[](i); // sizeof(item) == 0xC
      if (item.cell_form_id == cell->formID)
         return item.child_grup_offset;
   }
   return 0;
}

// LE: 445110 // separate subroutine from the above, taking a reference type

      if (item.cell_form_id == cell.formID)

   }
   return 0;
}

// LE: 446DA0
bool TESFile::SeekTo(uint32_t offset) {
   if (!this->unk10) {
      this->sub446130(&this->unk124, &this->unk20, 0, 0);
   }
   if (this->unk420) {
      this->unk420 = 0;
      this->unk424 = 0;
   }
   this->unk264 = offset;
   if (offset >= this->unk260) {
      this->currentForm = {};
      return false;
   }
   this->unk10->Unk_06(offset, *(uint32_t*)(0x01120D44));
   
   auto eax = this->unk10->unk38;
   if (eax == -1)
      eax = this->unk10->unk150;
   this->unk264 = eax;
   if (eax == -1) {
      FormatMessageA(0x1300, 0, GetLastError(), 0x400, &esp0C, 0, 0);
      //
      // presumably a logging call was #ifdef'd out
      //
      LocalFree(&esp0C);
      return false;
   }
   this->currentForm = {};
   this->EnsureIsInRecord();
   return true;
}

// LE: 446010
bool TESFile::SeekToRecordOf(const TESForm* form) {
   if (!form)
      return false;
   if (!this->unk298)
      return false;
   if (!this->unk10)
      return false;
   if (this->unk10->unk2C == 0)
      return false;
   
   if (form->FindInFileFast(this)) // virtual 0x0C // documented below
      return true;
   
   this->sub445F40(/*bool ensureInRecord=*/true);
   while (this->currentForm.signature == 'TES4')
      this->AdvanceToNextRecordOrGroup();
   
   uint32_t subject_id = form->formID;
   uint32_t subject_cc = FormTypeToSignatureList[form->GetSavedFormType()].signature;
   if (!form->GetSavedFormType())
      return false;
   
   bool result = false; // esp13
   bool bl     = true;  // bl
   do {
      if (!bl)
         return result;
      if (this->currentForm.signature == 'GRUP') {
         if (!form->BelongsInGroup(&this->currentForm, true, false)) {
            this->AdvanceToFirstRecordInGroup();
            continue;
         }
      }
      if (this->currentForm.formID == subject_id) {
         bl     = false;
         result = true;
         continue;
      }
      this->AdvanceToNextRecordOrGroup();
   } while (this->currentForm.signature);
   return result;
}

// LE: 4C0BC0
virtual bool TESObjectCELL::FindInFileFast(TESFile* file) override {
   if (!file) // edi
      return false;
   if (!file->IsMaster())
      return false;
   if (this->IsInterior()) { // this->unk2C & 1
      auto eax = this->sub4C08F0(file);
      if (eax)
         file->SeekTo(eax);
   } else {
      // at 4C0BF4
      if (TESWorldSpace* ebx = this->parentWorld) {
         int32_t x;
         int32_t y;
         this->GetCellCoordinates(&x, &y);
         ebx->sub4F2DC0(file, x, y);
      }
   }
   // at 4C0C23
   FormType ft = file->EnsureIsInRecord();
   return ft == FormType::Cell;
}

// LE: 4C08F0
uint32_t TESObjectCELL::sub4C08F0(TESFile* file) {
   if (!this->IsInterior())
      return 0;
   TESFile* edi = file->sub444890();
   if (!edi)
      edi = file;
   if (edi == this->GetSourceFile(0)) {
      if (!this->Isinterior())
         return 0;
      if (!this->loadedData)
         return 0;
      //
      // Cached by `TESObjectCELL::sub4C0960`, called when loading the `DATA` 
      // subrecord of an interior cell. This is the cell's offset within the 
      // file that originally defined it.
      //
      return this->loadedData->unk5C;
   }
   return file->OffsetOfInteriorCellRecordBody(*this); // documented above
}
